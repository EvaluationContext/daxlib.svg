=== "Definition"

    ```dax
    function 'DaxLib.SVG.Viz.Bars' =
    		(
    			axisRef: ANYREF EXPR,
    			measureRef: NUMERIC EXPR,
    			color: STRING = BLANK(),
    			minMarkColor: STRING = BLANK(),
    			maxMarkColor: STRING = BLANK(),
    			showAxis: BOOLEAN = FALSE(),
    			width: INT64 = 120,
    			height: INT64 = 48
    		) =>
    
    			VAR _Marks =
    				DaxLib.SVG.Compound.Bars(
    					0, 
                        0, 
                        width, 
                        height,
    					axisRef,
                        measureRef,
    					color, 
                        minMarkColor, 
                        maxMarkColor, 
                        showAxis
    				)
    
    			RETURN
    				IF( NOT ISBLANK( _Marks ), DaxLib.SVG.SVG( "100%", "100%", _Marks, "0 0 " & width & " " & height ) )
    ```
